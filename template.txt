// ===== SLAVE UART =====

// ===== VARIABEL SENSOR =====
int sensor1;
int sensor2;

void setup() {

  // UART
  Serial.begin(9600);

  // pinMode sensor
  pinMode(A0, INPUT);
  pinMode(7, INPUT);
}

void loop() {

  // ===== BACA SENSOR =====
  sensor1 = analogRead(A0);

  sensor2 = digitalRead(7);

  // ===== KIRIM DATA =====
  Serial.print("S1:");
  Serial.print(sensor1);

  Serial.print(",S2:");
  Serial.println(sensor2);

  delay(1000);
}
