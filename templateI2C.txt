// ===== SLAVE I2C =====

#include <Wire.h>

// alamat slave
#define SLAVE_ADDRESS 8

// variabel sensor
int sensor1;
int sensor2;

void setup() {

  // mulai I2C sebagai slave
  Wire.begin(SLAVE_ADDRESS);

  // saat master meminta data
  Wire.onRequest(kirimData);

  // setup sensor
  pinMode(A0, INPUT);
  pinMode(7, INPUT);
}

void loop() {

  // ===== BACA SENSOR =====
  sensor1 = analogRead(A0);

  sensor2 = digitalRead(7);

  delay(500);
}

// ===== FUNGSI KIRIM DATA =====
void kirimData() {

  String data = "";

  data += "S1:";
  data += sensor1;

  data += ",S2:";
  data += sensor2;

  Wire.write(data.c_str());
}


// ===== MASTER I2C =====

#include <Wire.h>

#define SLAVE_ADDRESS 8

void setup() {

  Wire.begin();

  Serial.begin(9600);
}

void loop() {

  // meminta data dari slave
  Wire.requestFrom(SLAVE_ADDRESS, 32);

  while(Wire.available()) {

    char c = Wire.read();

    Serial.print(c);
  }

  Serial.println();

  delay(1000);
}
