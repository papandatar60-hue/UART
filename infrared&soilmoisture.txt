// ===== SLAVE UART TINKERCAD =====

#include <SoftwareSerial.h>

// RX, TX
SoftwareSerial mySerial(2, 3);

int irPin = 7;
int soilPin = A0;

int irValue;
int soilValue;

void setup() {

  Serial.begin(9600);

  mySerial.begin(9600);

  pinMode(irPin, INPUT);
}

void loop() {

  // baca sensor
  irValue = digitalRead(irPin);

  soilValue = analogRead(soilPin);

  // kirim UART
  mySerial.print("IR:");

  if(irValue == LOW){
    mySerial.print("Ada");
  }
  else{
    mySerial.print("Tidak");
  }

  mySerial.print(",Soil:");
  mySerial.println(soilValue);

  delay(1000);
}

// ===== MASTER UART TINKERCAD =====

#include <SoftwareSerial.h>

// RX, TX
SoftwareSerial mySerial(2, 3);

String dataMasuk;

void setup() {

  Serial.begin(9600);

  mySerial.begin(9600);
}

void loop() {

  if(mySerial.available()) {

    dataMasuk = mySerial.readStringUntil('\n');

    Serial.print("Data: ");
    Serial.println(dataMasuk);
  }
}
