// SLAVE UART

int rainPin = A0;

int trigPin = 9;
int echoPin = 10;

long duration;
int distance;
int rainValue;

void setup() {
  Serial.begin(9600);
  
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
}

void loop() {

  // ===== SENSOR HUJAN =====
  rainValue = analogRead(rainPin);

  // ===== ULTRASONIK =====
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  duration = pulseIn(echoPin, HIGH);

  distance = duration * 0.034 / 2;

  // ===== KIRIM DATA UART =====
  Serial.print("Hujan:");
  Serial.print(rainValue);

  Serial.print(",Jarak:");
  Serial.println(distance);

  delay(1000);
}

// MASTER UART

String dataMasuk;

void setup() {
  Serial.begin(9600);
}

void loop() {

  if (Serial.available()) {

    dataMasuk = Serial.readStringUntil('\n');

    Serial.print("Data diterima: ");
    Serial.println(dataMasuk);
  }
}
